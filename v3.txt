function Library:GetTransparency(obj)
	if IsA(obj, "Frame") then
		return { "BackgroundTransparency" }
	elseif IsA(obj, "TextLabel") or IsA(obj, "TextButton") then
		return { "TextTransparency", "BackgroundTransparency" }
	elseif IsA(obj, "ImageLabel") or IsA(obj, "ImageButton") then
		return { "BackgroundTransparency", "ImageTransparency" }
	elseif IsA(obj, "ScrollingFrame") then
		return { "BackgroundTransparency", "ScrollBarImageTransparency" }
	elseif IsA(obj, "TextBox") then
		return { "TextTransparency", "BackgroundTransparency" }
	elseif IsA(obj, "UIStroke") then
		return { "Transparency" }
	end

	return nil
end

function Library:Tween(Object, Properties, Info)
	local tween = TweenService:Create(
		Object,
		Info or TweenInfo.new(Library.TweeningSpeed, Library.EasingStyle, Enum.EasingDirection.InOut, 0, false, 0),
		Properties
	)
	tween:Play()

	return tween
end

function Library:Fade(obj, prop, vis, speed)
	if not (obj and prop) then
		return
	end

	local OldTransparency = obj[prop]
	obj[prop] = vis and 1 or OldTransparency

	local Tween = Library:Tween(
		obj,
		{ [prop] = vis and OldTransparency or 1 },
		TweenInfo.new(speed or Library.TweeningSpeed, Library.EasingStyle, Enum.EasingDirection.InOut, 0, false, 0)
	)

	Library:Connection(Tween.Completed, function()
		if not vis then
			Wait()
			obj[prop] = OldTransparency
		end
	end)

	return Tween
end

--[[
		Convert AbsolutePosition into a Position usable under an IgnoreGuiInset ScreenGui.
		Pointer coords include the top-bar inset; AbsolutePosition does not, so writing it back shifts the UI up.
	]]
local function absoluteToGuiOffset(guiObject)
	local abs = guiObject.AbsolutePosition
	local gui = guiObject:FindFirstAncestorWhichIsA("ScreenGui")
	if gui and gui.IgnoreGuiInset then
		return abs + GetGuiInset(GuiService)
	end
	return abs
end

---Press: left mouse or touch; shared by drag, slider, and NewColor3 picker.
---@param input InputObject
---@return boolean
local function isPressInput(input)
	local inputType = input.UserInputType
	return inputType == Enum.UserInputType.MouseButton1 or inputType == Enum.UserInputType.Touch
end

---Move: mouse movement or finger slide. On mobile, InputChanged uses Touch instead of MouseMovement.
---@param input InputObject
---@return boolean
local function isMoveInput(input)
	local inputType = input.UserInputType
	return inputType == Enum.UserInputType.MouseMovement or inputType == Enum.UserInputType.Touch
end

function Library:Resizify(Parent)
	local Resizing = Library:Create("TextButton", {
		Position = NewUDim2(1, -10, 1, -10),
		BorderColor3 = NewRGB(0, 0, 0),
		Size = NewUDim2(0, 10, 0, 10),
		BorderSizePixel = 0,
		BackgroundColor3 = NewRGB(255, 255, 255),
		Parent = Parent,
		BackgroundTransparency = 1,
		Text = "",
		Active = true,
	})

	local IsResizing = false
	local Size
	local InputLost
	local ParentSize = Parent.Size

	Connect(Resizing.InputBegan, function(input)
		if isPressInput(input) then
			IsResizing = true
			InputLost = input.Position
			Size = Parent.Size
		end
	end)

	Connect(Resizing.InputEnded, function(input)
		if isPressInput(input) then
			IsResizing = false
		end
	end)

	Library:Connection(UserInputService.InputChanged, function(input)
		if IsResizing and isMoveInput(input) then
			Parent.Size = NewUDim2(
				Size.X.Scale,
				Clamp(Size.X.Offset + (input.Position.X - InputLost.X), ParentSize.X.Offset, Camera.ViewportSize.X),
				Size.Y.Scale,
				Clamp(Size.Y.Offset + (input.Position.Y - InputLost.Y), ParentSize.Y.Offset, Camera.ViewportSize.Y)
			)
		end
	end)

	Library:Connection(UserInputService.InputEnded, function(input)
		if isPressInput(input) then
			IsResizing = false
		end
	end)
end

function Library:Hovering(Object)
	if Type(Object) == "table" then
		local Pass = false
		for i = 1, #Object do
			if Library:Hovering(Object[i]) then
				Pass = true
				return Pass
			end
		end
	else
		-- GetMouseLocation follows the last touch; GetMouse() often sticks at a stale position on mobile
		local loc = MouseLocation()
		local inset = GetGuiInset(GuiService)
		local pointerX = loc.X - inset.X
		local pointerY = loc.Y - inset.Y
		local y_cond = Object.AbsolutePosition.Y <= pointerY
			and pointerY <= Object.AbsolutePosition.Y + Object.AbsoluteSize.Y
		local x_cond = Object.AbsolutePosition.X <= pointerX
			and pointerX <= Object.AbsolutePosition.X + Object.AbsoluteSize.X

		return (y_cond and x_cond)
	end
end

function Library:Draggify(Parent, extraHandles, allowOffscreen, onDragEnded)
	local Dragging = false
	local InitialMousePosition
	local InitialOffset

	-- Mobile touch never lands on Frames with Active=false
	Parent.Active = true

	local function beginDrag(input)
		local offset = absoluteToGuiOffset(Parent)
		Parent.AnchorPoint = NewVector2(0, 0)
		Parent.Position = NewUDim2(0, offset.X, 0, offset.Y)
		Dragging = true
		InitialMousePosition = input.Position
		InitialOffset = offset
	end

	local function endDrag()
		if not Dragging then
			return
		end
		Dragging = false
		if onDragEnded then
			pcall(onDragEnded, Parent)
		end
	end

	local function bindHandle(handle)
		if not handle or not IsA(handle, "GuiObject") then
			return
		end
		Connect(handle.InputBegan, function(input)
			if isPressInput(input) then
				beginDrag(input)
			end
		end)
	end

	bindHandle(Parent)
	-- true: whole NewRect is draggable (watermark / Notification / Keybind patches without buttons)
	if extraHandles == true then
		Library:Connection(UserInputService.InputBegan, function(input)
			if isPressInput(input) and Parent.Visible and Parent.Parent and Library:Hovering(Parent) then
				beginDrag(input)
			end
		end)
	elseif Type(extraHandles) == "table" then
		for i = 1, #extraHandles do
			bindHandle(extraHandles[i])
		end
	end

	Library:Connection(UserInputService.InputChanged, function(input)
		if not (Dragging and isMoveInput(input)) then
			return
		end
		local delta = input.Position - InitialMousePosition
		local newX = InitialOffset.X + delta.X
		local newY = InitialOffset.Y + delta.Y
		-- Watermark-style patches stay clamped in the viewport; the main window may leave the screen
		if not allowOffscreen then
			local viewport = Camera.ViewportSize
			local size = Parent.AbsoluteSize
			newX = Clamp(newX, 0, Max(0, viewport.X - size.X))
			newY = Clamp(newY, 0, Max(0, viewport.Y - size.Y))
		end
		Parent.AnchorPoint = NewVector2(0, 0)
		Parent.Position = NewUDim2(0, newX, 0, newY)
	end)

	-- End drag on release even if the finger left the control
	Library:Connection(UserInputService.InputEnded, function(input)
		if isPressInput(input) then
			endDrag()
		end
	end)
end

function Library:Convert(str)
	local Values = {}

	for Value in string.gmatch(str, "[^,]+") do
		Insert(Values, tonumber(Value))
	end

	if #Values == 4 then
		return unpack(Values)
	else
		return
	end
end

function Library:Lerp(start, finish, t)
	t = t or 1 / 8

	return start * (1 - t) + finish * t
end

--[[
		Restore a config key string into an EnumItem.
		Use the already-defined Insert so missing locals cannot skip an entire Keybind during load.
		@param enum EnumItem|string|nil
		@return EnumItem|"NONE"
	]]
function Library:ConvertEnum(enum)
	if TypeOf(enum) == "EnumItem" then
		return enum
	end
	if enum == nil or enum == "NONE" or enum == "nil" or enum == "" then
		return "NONE"
	end
	if Type(enum) ~= "string" then
		return "NONE"
	end

	local enumParts = {}
	for part in string.gmatch(enum, "[%w_]+") do
		Insert(enumParts, part)
	end

	-- For "Enum.KeyCode.E", the first segment is Enum; index from the enum type name
	local startIndex = 1
	if enumParts[1] == "Enum" then
		startIndex = 2
	end
	if enumParts[startIndex] == nil then
		return "NONE"
	end

	local enumTable = Enum
	for i = startIndex, #enumParts do
		local nextItem = enumTable[enumParts[i]]
		if nextItem == nil then
			return "NONE"
		end
		enumTable = nextItem
	end

	if TypeOf(enumTable) == "EnumItem" then
		return enumTable
	end
	return "NONE"
end

function Library:ConvertHex(NewColor3, alpha)
	local r = Floor(NewColor3.R * 255)
	local g = Floor(NewColor3.G * 255)
	local b = Floor(NewColor3.B * 255)
	local a = alpha and Floor(alpha * 255) or 255
	return Format("#%02X%02X%02X%02X", r, g, b, a)
end

function Library:ConvertFromHex(hexString)
	hexString = hexString:gsub("#", "")
	local r = tonumber(hexString:sub(1, 2), 16) / 255
	local g = tonumber(hexString:sub(3, 4), 16) / 255
	local b = tonumber(hexString:sub(5, 6), 16) / 255
	local a = tonumber(hexString:sub(7, 8), 16) and tonumber(hexString:sub(7, 8), 16) / 255 or 1
	return NewColor3(r, g, b), a
end

--[[
		Restore a config NewColor3 value into Color3.
		Accepts FromHex strings, {R,G,B} tables, Color3, and { Color = ..., Transparency = ... } payloads.
		@param value any
		@return Color3|nil
	]]
function Library:ParseConfigColor(value)
	if TypeOf(value) == "Color3" then
		return value
	end
	if Type(value) == "string" then
		local hexValue = value:gsub("#", "")
		if #hexValue < 6 then
			return nil
		end
		hexValue = hexValue:sub(1, 6)
		local ok, parsed = pcall(FromHex, hexValue)
		if ok and TypeOf(parsed) == "Color3" then
			return parsed
		end
		return nil
	end
	if Type(value) ~= "table" then
		return nil
	end
	if value.Color ~= nil then
		return Library:ParseConfigColor(value.Color)
	end
	if Type(value.R) == "number" and Type(value.G) == "number" and Type(value.B) == "number" then
		return NewColor3(value.R, value.G, value.B)
	end
	return nil
end

--[[
		Take the config short name from a listfiles full path (no directory, no .cfg).
		Normalize slashes so Windows UNC paths do not bake the folder into the name and break Load on rejoin.
		@param file string
		@return string
	]]
local function GetConfigBasename(file)
	local name = tostring(file or ""):gsub("\\", "/")
	name = name:match("([^/]+)$") or name
	name = name:gsub("%.cfg$", "")
	name = name:match("^%s*(.-)%s*$") or name
	return name
end

--[[
		Normalize a textbox/dropdown value into a readable/writable config path.
		@param name string
		@return string|nil, string|nil Short name and full path; both nil when the name is empty
	]]
local function ResolveConfigPath(name)
	local basename = GetConfigBasename(name)
	if basename == "" then
		return nil, nil
	end
	return basename, Library.Directory .. "/configs/" .. basename .. ".cfg"
end

--[[
		Whether the Keybind key is currently down (keep Active on Hold load if still held).
		@param key EnumItem|string
		@return boolean
	]]
local function isKeybindKeyDown(key)
	if TypeOf(key) ~= "EnumItem" then
		return false
	end
	if key.EnumType == Enum.KeyCode then
		return UserInputService:IsKeyDown(key)
	end
	if key.EnumType == Enum.UserInputType then
		return UserInputService:IsMouseButtonPressed(key)
	end
	return false
end

--[[
		Serialize a runtime Keybind Key into a JSON-safe string.
		@param key EnumItem|string|nil
		@return string
	]]
local function serializeKeybindKey(key)
	if key == nil or key == "NONE" or key == "nil" then
		return "NONE"
	end
	if TypeOf(key) == "EnumItem" then
		return tostring(key)
	end
	return tostring(key)
end

		and isFiniteNumber(value.YScale)
		and isFiniteNumber(value.YOffset)
		and isFiniteNumber(value.AnchorX)
		and isFiniteNumber(value.AnchorY)
end

function Library:SaveOverlayPositions(overlays)
	local positions = {}
	for name, guiObject in overlays do
		local position = serializeOverlayPosition(guiObject)
		if position then
			positions[name] = position
		end
	end

	local encodeOk, encoded = pcall(HttpService.JSONEncode, HttpService, positions)
	if not encodeOk then
		return false
	end

	local writeOk = pcall(writefile, OVERLAY_POSITIONS_FILE, encoded)
	return writeOk
end

function Library:LoadOverlayPositions(overlays)
	if not isfile(OVERLAY_POSITIONS_FILE) then
		return false
	end

	local readOk, contents = pcall(readfile, OVERLAY_POSITIONS_FILE)
	if not readOk or Type(contents) ~= "string" then
		return false
	end

	local decodeOk, positions = pcall(HttpService.JSONDecode, HttpService, contents)
	if not decodeOk or Type(positions) ~= "table" then
		return false
	end

	for name, guiObject in overlays do
		local position = positions[name]
		if guiObject and isValidOverlayPosition(position) then
			guiObject.AnchorPoint = NewVector2(position.AnchorX, position.AnchorY)
			guiObject.Position = NewUDim2(
				position.XScale,
				position.XOffset,
				position.YScale,
				position.YOffset
			)
		end
	end

	return true
end

local ConfigHolder
function Library:UpdateConfigList()
	if not ConfigHolder then
		return
	end

	local List = {}
	local ok, files = pcall(listfiles, Library.Directory .. "/configs")
	if ok and Type(files) == "table" then
		for _, file in files do
			local configName = GetConfigBasename(file)
			if configName ~= "" then
				List[#List + 1] = configName
			end
		end
	end

	ConfigHolder.RefreshOptions(List)
end

		Items.InlineColorPicker.BackgroundColor3 = FromHSV(h, s, v)

--[[
		Serialize current Flags to JSON. Keybind Keys are always Enum strings; skip config-name and Mode dropdown Flags.
		@return string|nil
	]]
function Library:GetConfig()
	local Config = {}

	for Idx, Value in Flags do
		if isSkippedConfigFlag(Idx) then
			continue
		end
		if Type(Value) == "table" and Value.Color ~= nil then
			-- Do not use `if Transparency then`: default 0 is falsy and would write Color3 userdata into JSON
			local parsed = Library:ParseConfigColor(Value.Color)
			if parsed then
				Config[Idx] = {
					Transparency = Value.Transparency or 0,
					Color = parsed:ToHex(),
				}
			end
		elseif Type(Value) == "table" and (Value.key ~= nil or Value.Key ~= nil or Value.mode or Value.Mode) then
			Config[Idx] = {
				Mode = Value.mode or Value.Mode or "Toggle",
				Active = Value.active == true or Value.Active == true,
				Key = serializeKeybindKey(Value.key or Value.Key),
			}
		else
			local valueType = TypeOf(Value)
			if valueType == "boolean" or valueType == "number" or valueType == "string" or valueType == "table" then
				Config[Idx] = Value
			end
		end
	end

	local ok, encoded = pcall(HttpService.JSONEncode, HttpService, Config)
	if not ok then
		return nil
	end
	return encoded
end

--[[
		Restore config from JSON. Apply toggles/options before Keybinds; block Toggle/Keybind sync and Mode dropdowns from turning features off while loading.
		@param JSON string
		@return boolean
	]]
function Library:LoadConfig(JSON)
	if Type(JSON) ~= "string" or JSON == "" then
		return false
	end

	local decodeOk, Config = pcall(HttpService.JSONDecode, HttpService, JSON)
	if not decodeOk or Type(Config) ~= "table" then
		return false
	end

	local function isColorPayload(value)
		return Type(value) == "table" and value.Color ~= nil
	end

	--[[
			Keybind payload: has Mode/Key/Active. Exclude NewColor3 tables so Colorpickers are not treated as binds.
			@param value any
			@return boolean
		]]
	local function isKeybindPayload(value)
		if Type(value) ~= "table" then
			return false
		end
		if isColorPayload(value) then
			return false
		end
		return value.Mode ~= nil
			or value.mode ~= nil
			or value.Key ~= nil
			or value.key ~= nil
			or value.Active ~= nil
			or value.active ~= nil
	end

	--[[
			Derive Active from Mode: Always on, Hold only while key is down, Toggle uses the saved value.
			@param value table
			@return table
		]]
	local function normalizeKeybind(value)
		local mode = value.Mode or value.mode or "Toggle"
		local key = value.Key or value.key or "NONE"
		local active
		if mode == "Always" then
			active = true
		elseif mode == "Hold" then
			active = isKeybindKeyDown(Library:ConvertEnum(key))
		else
			local saved = value.Active
			if saved == nil then
				saved = value.active
			end
			active = saved == true
		end
		return {
			Mode = mode,
			Key = key,
			Active = active,
		}
	end

	local function applyFlag(idx, value)
		local setter = ConfigFlags[idx]
		if not setter then
			return
		end
		if isColorPayload(value) then
			local parsed = Library:ParseConfigColor(value)
			if parsed then
				setter(parsed, value.Transparency or 0)
				return
			end
		end
		setter(value)
	end

	-- Restore toggles/sliders/dropdowns first, then Keybinds; block Toggle/Keybind sync from rewriting the parent while loading
	Library.IsLoadingConfig = true
	pcall(function()
		for Idx, Value in Config do
			if isSkippedConfigFlag(Idx) or isKeybindPayload(Value) then
				continue
			end
			pcall(applyFlag, Idx, Value)
		end
		for Idx, Value in Config do
			if isSkippedConfigFlag(Idx) or not isKeybindPayload(Value) then
				continue
			end
			pcall(applyFlag, Idx, normalizeKeybind(Value))
		end
	end)
	Library.IsLoadingConfig = false

	return true
end

function Library:Round(num, float)
	local Multiplier = 1 / (float or 1)
	return Floor(num * Multiplier + 0.5) / Multiplier
end

--[[
		Update global font sizes: body uses size, control titles are size+2 (default 14 / 16 gap).
		Walk FontSizeRegistry so the Font Size slider refreshes the whole UI immediately.
		@param size number Body text size
	]]
function Library:SetFontSize(size)
	Library.TextSize = size
	Library.TextSizeHeader = size + 2

	for _, entry in Library.FontSizeRegistry do
		entry.Instance.TextSize = entry.Kind == "header" and Library.TextSizeHeader or Library.TextSize
	end
end

--[[
		Update the menu font and re-apply it to every Library-created text instance so the
		Menu Font dropdown restyles the whole open UI immediately (mirrors SetFontSize).
		@param font Font
	]]
function Library:SetFont(font)
	Library.Font = font

	-- Some registered instances may have been Destroy()ed by a later RefreshOptions, so guard each write.
	for _, instance in Library.FontRegistry do
		pcall(function()
			instance.FontFace = font
		end)
	end
end

---Close the currently open floating element (Dropdown panel, etc.).
---Dropdown stores the open Cfg directly; older callers may pass an array of Cfgs.
function Library:CloseElement()
	local element = Library.OpenElement
	if not element then
		return
	end

	-- Single open element (Dropdown sets OpenElement = Cfg)
	if element.SetVisible then
		if not element.Ignore then
			element.Open = false
			element.SetVisible(false)
		end
		Library.OpenElement = {}
		return
	end

	for i = 1, #element do
		local Data = element[i]

		if Data and not Data.Ignore and Data.SetVisible then
			Data.Open = false
			Data.SetVisible(false)
		end
	end

	Library.OpenElement = {}
end

function Library:Create(instance, options)
	local ins = NewInstance(instance)

	for prop, value in options do
		ins[prop] = value
	end

	-- Register text instances that follow global font size (body / header / overlay patches) for the Font Size slider.
	if options.TextSize ~= nil then
		if options.TextSize == Library.TextSize then
			Insert(Library.FontSizeRegistry, { Instance = ins, Kind = "body" })
		elseif options.TextSize == Library.TextSizeHeader then
			Insert(Library.FontSizeRegistry, { Instance = ins, Kind = "header" })
		end
	end

	-- Register text instances that carry a FontFace so the Menu Font dropdown can restyle them live.
	if options.FontFace ~= nil then
		Insert(Library.FontRegistry, ins)
	end

	if ins == "TextButton" then
		ins["AutoButtonColor"] = false
		ins["Text"] = ""
	end

	return ins
end

function Library:Unload()
	if Library.Items then
		Library.Items:Destroy()
		Library.Items = nil
	end

	if Library.Other then
		Library.Other:Destroy()
		Library.Other = nil
	end

	for index, connection in Library.Connections do
		Disconnect(connection)
		Library.Connections[index] = nil
	end

	Library = nil
end

---Format watermark date like constant pf.lua (`%B. %d, %Y`).
---@return string
local function formatWatermarkDate()
	return Date("%B. %d, %Y")
end

-- Spawn-style cache for Stats tree walks used by the watermark periodic refresh path.
local watermarkHeartbeatStat = nil
local watermarkDataPingStat = nil

---Refresh cached Stats instances used by watermark FPS / ping (not every frame).
local function refreshWatermarkStatCache()
	local workspaceStats = FindFirstChild(Stats, "Workspace") -- hot-ok cache refresh
	watermarkHeartbeatStat = workspaceStats and FindFirstChild(workspaceStats, "Heartbeat") or nil -- hot-ok cache refresh
	local networkStats = FindFirstChild(Stats, "Network") -- hot-ok cache refresh
	local serverStats = networkStats and FindFirstChild(networkStats, "ServerStatsItem") or nil -- hot-ok cache refresh
	watermarkDataPingStat = serverStats and FindFirstChild(serverStats, "Data Ping") or nil -- hot-ok cache refresh
end

---Read current FPS (prefer cached Stats.Workspace.Heartbeat).
---@return number
local function getWatermarkFps()
	local heartbeat = watermarkHeartbeatStat
	if not heartbeat or not heartbeat.Parent then
		refreshWatermarkStatCache()
		heartbeat = watermarkHeartbeatStat
	end
	if not heartbeat then
		return 0
	end
	local ok, fps = pcall(function()
		return Floor(heartbeat:GetValue() + 0.5)
	end)
	return ok and fps or 0
end

---Read current network ping in milliseconds.
---@return number
local function getWatermarkPing()
	if lp and lp.GetNetworkPing then
		local ok, ping = pcall(function()
			return Floor(lp:GetNetworkPing() * 1000 + 0.5)
		end)
		if ok then
			return ping
		end
	end
	local dataPing = watermarkDataPingStat
	if not dataPing or not dataPing.Parent then
		refreshWatermarkStatCache()
		dataPing = watermarkDataPingStat
	end
	if not dataPing then
		return 0
	end
	local ok, ping = pcall(function()
		return Floor(dataPing:GetValue() + 0.5)
	end)
	return ok and ping or 0
end

---Replace `%needle%` with a live value using a plain search so `%fps%` is not parsed as a Lua pattern.
---@param haystack string Watermark format text
---@param needle string Token without percent signs, e.g. scriptname
---@param replacement string Live value for that token
---@return string
local function replaceWatermarkToken(haystack, needle, replacement)
	local token = "%" .. needle .. "%"
	replacement = tostring(replacement or "")
	local pieces = {}
	local cursor = 1
	while true do
		local found = StringFind(haystack, token, cursor, true)
		if not found then
			pieces[#pieces + 1] = Sub(haystack, cursor)
			break
		end
		if found > cursor then
			pieces[#pieces + 1] = Sub(haystack, cursor, found - 1)
		end
		pieces[#pieces + 1] = replacement
		cursor = found + #token
	end
	return Concat(pieces)
end

---Expand Watermark format tokens. Missing tokens stay omitted because the user deleted them from the textbox.
---@param format string|nil User format, default WATERMARK_DEFAULT_FORMAT
---@param scriptName string|nil Menu title used by %scriptname%
---@return string
local function buildWatermarkText(format, scriptName)
	if Type(format) ~= "string" or format == "" then
		format = WATERMARK_DEFAULT_FORMAT
	end
	if Type(scriptName) ~= "string" or scriptName == "" then
		scriptName = WATERMARK_DEFAULT_TITLE
	end
	local text = format
	text = replaceWatermarkToken(text, "scriptname", scriptName)
	text = replaceWatermarkToken(text, "date", formatWatermarkDate())
	text = replaceWatermarkToken(text, "fps", tostring(getWatermarkFps()) .. " fps")
	text = replaceWatermarkToken(text, "ping", tostring(getWatermarkPing()) .. " ms")
	return text
end

---Measure overlay text width with the given horizontal padding.
---@param text string
---@param padding number Extra pixels added to text bounds
---@return number
local function getOverlayTextWidth(text, padding)
	text = text or ""
	padding = padding or 0
	local ok, bounds = pcall(function()
		local params = NewInstance("GetTextBoundsParams")
		params.Text = text
		params.Font = Library.Font
		params.Size = Library.TextSize
		params.Width = 10000
		return TextService:GetTextBoundsAsync(params)
	end)
	if ok and bounds then
		return bounds.X + padding
	end
	local legacyBounds =
		TextService:GetTextSize(text, Library.TextSize, Enum.Font.SourceSans, NewVector2(10000, WATERMARK_HEIGHT))
	return legacyBounds.X + padding
end

	function Cfg.ToggleMenu(bool)
		if Cfg.Tweening then
			return
		end

		-- DropdownElements live under Library.Items (outside Window); close before Window hide
		if not bool then
			Library:CloseElement()
		end

		Cfg.Tweening = true

		if bool then
			Items.Window.Visible = true
		end

		local Children = GetDescendants(Items.Window)
		Insert(Children, Items.Window)

		local Tween
		for ci = 1, #Children do
			local obj = Children[ci]
			local Index = Library:GetTransparency(obj)

			if not Index then
				continue
			end

			if Type(Index) == "table" then
				for _, prop in Index do
					Tween = Library:Fade(obj, prop, bool)
				end
			else
				Tween = Library:Fade(obj, Index, bool)
			end
		end

		Library:Connection(Tween.Completed, function()
			Cfg.Tweening = false
			Items.Window.Visible = bool
		end)
	end

	function Cfg.ChangeTitle(text)
		if Type(text) ~= "string" or text == "" then
			text = Cfg.Name or WATERMARK_DEFAULT_TITLE
		end

		-- %scriptname% tracks the Menu title

			Cfg.SetWatermarkText(buildWatermarkText(Cfg.WatermarkFormat, Cfg.Name))
		end
	end

		if Cfg.SetWatermarkText then
			Cfg.SetWatermarkText(buildWatermarkText(Cfg.WatermarkFormat, Cfg.Name))
		end
	end

	function Cfg.ToggleNotify(bool)
		Items.Notify.Visible = bool
	end

	function Cfg.ToggleKeybindList(bool)
		Items.Keybind_List.Visible = bool
		if Items.Keybind_Holder then
			Items.Keybind_Holder.Visible = bool
		end
	end

	return setmetatable(Cfg, Library)

---Player List: roster + priority. View pins CurrentCamera to the selected opponent Humanoid for spectating.
---@param properties table|nil Reserved; PlayerList currently ignores the properties table
---@return table PlayerList control (includes GetPriority / ModifyPriority)
function Library:PlayerList(properties)
	local selectionRequest = 0
	local Cfg = {
		Players = {},
		Items = {},

		Selected,
	}

	local Colors = {
		["Neutral"] = themes.preset.text_color,
		["Friendly"] = NewRGB(38, 89, 140),
		["Enemy"] = NewRGB(140, 0, 2),
	}

	-- View must rewrite CameraSubject every frame: most games reclaim the camera after death/respawn.
	local viewBindName = "MoonLuaPlayerListView_" .. tostring(Cfg)
	local viewBound = false
	local viewHadTarget = false
	local viewCachedPlayer = nil
	local viewCachedHumanoid = nil
	local viewCharAddedConn = nil
	local viewCharRemovingConn = nil

	---Online opponent for the selected row; local player and left-server entries are never spectated.
	---@return Player|nil
	local function GetSelectedPlayer()
		local selected = Cfg.Selected
		if not selected then
			return nil
		end
		local player = selected.Player
		if TypeOf(player) ~= "Instance" or not IsA(player, "Player") then
			return nil
		end
		if player.Parent ~= Players or player == Players.LocalPlayer then
			return nil
		end
		return player
	end

	---Bind Character Humanoid once for the spectate target (Spawn-style cache).
	---@param player Player|nil
	local function BindViewTarget(player)
		if viewCharAddedConn then
			Disconnect(viewCharAddedConn)
			viewCharAddedConn = nil
		end
		if viewCharRemovingConn then
			Disconnect(viewCharRemovingConn)
			viewCharRemovingConn = nil
		end
		viewCachedPlayer = player
		viewCachedHumanoid = nil
		if not player then
			return
		end
		local function cacheHumanoid(character)
			viewCachedHumanoid = character and FindFirstChildOfClass(character, "Humanoid") or nil -- hot-ok Spawn bind
		end
		cacheHumanoid(player.Character)
		viewCharAddedConn = Connect(player.CharacterAdded, cacheHumanoid)
		viewCharRemovingConn = Connect(player.CharacterRemoving, function()
			viewCachedHumanoid = nil
		end)
	end

	---Return the camera to the local Humanoid when spectating ends or the target is lost.
	local function RestoreLocalCamera()
		local character = Players.LocalPlayer.Character
		local humanoid = character and FindFirstChildOfClass(character, "Humanoid") -- hot-ok restore once
		local cam = Workspace.CurrentCamera
		if humanoid and cam then
			cam.CameraSubject = humanoid
		end
	end

	---Stop the spectate loop; restore the camera only if spectating actually ran, so Toggle init Callback(false) does not steal the subject.
	local function StopView()
		local wasViewing = viewBound or viewHadTarget
		if viewBound then
			RunService:UnbindFromRenderStep(viewBindName)
			viewBound = false
		end
		BindViewTarget(nil)
		viewHadTarget = false
		if wasViewing then
			RestoreLocalCamera()
		end
	end

	---Keep overwriting CameraSubject for the selected opponent; switching rows needs no rebind because the loop reads Cfg.Selected.
	local function ApplyView()
		if viewBound then
			RunService:UnbindFromRenderStep(viewBindName)
			viewBound = false
		end
		-- Run after Roblox's camera update so CameraSubject is not overwritten in the same frame.
		RunService:BindToRenderStep(viewBindName, Enum.RenderPriority.Camera.Value + 1, function()
			local target = GetSelectedPlayer()
			local cam = Workspace.CurrentCamera
			if not target or not cam then
				-- Restore once after the target leaves so an empty selection does not steal the local camera every frame.
				if viewHadTarget then
					RestoreLocalCamera()
					viewHadTarget = false
				end
				if viewCachedPlayer ~= nil then
					BindViewTarget(nil)
				end
				return
			end
			if viewCachedPlayer ~= target then
				BindViewTarget(target)
			end
			local humanoid = viewCachedHumanoid
			if not humanoid or not humanoid.Parent then
				local character = target.Character
				humanoid = character and FindFirstChildOfClass(character, "Humanoid") or nil -- hot-ok delayed Humanoid retry
				viewCachedHumanoid = humanoid
			end
			if not humanoid then
				return
			end
			viewHadTarget = true
			cam.CameraSubject = humanoid
		end)
		viewBound = true
	end

	if self.Items.LeftScroll then
		self.Items.LeftScroll:Destroy()
	end
	if self.Items.RightScroll then
		self.Items.RightScroll:Destroy()
	end

	local Items = Cfg.Items
	do
		Library:Create("UIListLayout", {
			FillDirection = Enum.FillDirection.Horizontal,
			HorizontalFlex = Enum.UIFlexAlignment.Fill,
			Parent = self.Items.Page,
			Padding = NewUDim(0, 4),
			SortOrder = Enum.SortOrder.LayoutOrder,
			VerticalFlex = Enum.UIFlexAlignment.Fill,
		})

		Items.Column = Library:Create("Frame", {
			Parent = self.Items.Page,
			BackgroundTransparency = 1,

			BorderColor3 = NewRGB(0, 0, 0),

			BorderSizePixel = 0,

		})

			local Old = Cfg.Selected

			if Old then
				Old.Name.TextColor3 = themes.preset.text_color
				Old.UserId.TextColor3 = themes.preset.text_color
			end

			SeperateData.Name.TextColor3 = themes.preset.accent
			SeperateData.UserId.TextColor3 = themes.preset.accent
			Cfg.Selected = SeperateData
			selectionRequest += 1
			local request = selectionRequest
			Items.ImageProfile.Image = ""

			Items.NameLabel.Set("Name: " .. player.Name)
			Items.UserIdLabel.Set("UserId: " .. player.UserId)
			Items.DisplayNameLabel.Set("DisplayName: " .. player.DisplayName)

			-- Thumbnail requests may yield and finish after a newer selection.
			Spawn(function()
				local success, thumbnail = pcall(function()
					return Players:GetUserThumbnailAsync(
						player.UserId,
						Enum.ThumbnailType.AvatarThumbnail,
						Enum.ThumbnailSize.Size150x150
					)
				end)
				if
					success
					and request == selectionRequest
					and Cfg.Selected == SeperateData
					and Items.ImageProfile.Parent
				then
					Items.ImageProfile.Image = thumbnail
				end
			end)
			Wait()
			if request == selectionRequest and Cfg.Selected == SeperateData then
				Items.PriorityDropdown.Set(SeperateData.Priority.Text)
			end
		end)

		Cfg.Players[player.Name] = SeperateData
	end

	function Cfg.ModifyPriority(Priority, Items)
		local Path = Items or Cfg.Selected
		if Path then
			Path.Priority.Text = Priority
			Path.Priority.TextColor3 = Colors[Priority]
		end
	end

	function Cfg.GetPriority(Player)
		return Cfg.Players[Player].Priority.Text
	end

	local function applyPlayerSearch()
		local query = string.lower(Items.Search.Text):gsub("^%s+", ""):gsub("%s+$", "")
		for _, player in Players:GetPlayers() do
			local row = Cfg.Players[player.Name]
			if not row then
				continue
			end
			if query == "" then
				row.Background.Visible = true
				continue
			end
			local nameHit = string.find(string.lower(player.Name), query, 1, true) ~= nil
			local displayHit = string.find(string.lower(player.DisplayName), query, 1, true) ~= nil
			local idHit = string.find(tostring(player.UserId), query, 1, true) ~= nil
			row.Background.Visible = nameHit or displayHit or idHit
		end
	end

	for _, player in Players:GetPlayers() do
		Cfg.AddPlayer(player)
	end
	applyPlayerSearch()

	Connect(Items.Refresh.MouseButton1Click, function()
		for _, player in Players:GetPlayers() do
			if not Cfg.Players[player.Name] then
				Cfg.AddPlayer(player)
			end
		end
		applyPlayerSearch()
	end)

	Connect(GetPropertyChangedSignal(Items.Search, "Text"), applyPlayerSearch)

	return setmetatable(Cfg, Library)
end

	--[[
		Hide the expand arrow when a folding toggle has no real child controls (only Colorpicker / Keybind on the right).
	]]
	function Cfg.RefreshFolding()
		if not Items.GroupElements or not Items.ArrowHolder then
			return
		end
		local childCount = 0
		local groupChildren = GetChildren(Items.GroupElements)
		for gi = 1, #groupChildren do
			local child = groupChildren[gi]
			if not IsA(child, "UIListLayout") and not IsA(child, "UIPadding") then
				childCount += 1
			end
		end
		if childCount > 0 then
			return
		end
		Items.ArrowHolder.Visible = false
		if Items.Group then
			Items.Group.Visible = false
			Items.Group.Parent = Library.Other
		end
		Cfg.Folding = false
		Cfg.DecoupleFold = false
		Cfg.Expanded = false
	end

	function Cfg.SetExpanded(visible)
		if not Cfg.Folding then
			return
		end
		Cfg.Expanded = visible == true
		if Cfg.Expanded and Items.Object.Parent then
			Items.Group.Parent = Items.FoldContainer
			Items.Group.Visible = true
			Library:Tween(Items.Arrow, { Rotation = 180 })
			Library:Tween(Items.Fade, { BackgroundTransparency = 1 })
		else
			Items.Group.Visible = false
			Items.Group.Parent = Library.Other
			Library:Tween(Items.Arrow, { Rotation = 0 })
			Items.Fade.BackgroundTransparency = 0
		end
	end

	function Cfg.Set(bool)
		Cfg.Enabled = bool
		Flags[Cfg.Flag] = bool
		Cfg.Callback(bool)

		Library:Tween(Items.Accent, { BackgroundColor3 = bool and themes.preset.accent or themes.preset.inline })

		if Cfg.Folding and not Cfg.DecoupleFold then
			Items.Group.Visible = bool
			Items.Group.Parent = bool and Items.FoldContainer or Library.Other
			Items.Fade.BackgroundTransparency = 0

			Library:Tween(Items.Arrow or Items.Arrow, { Rotation = bool and 180 or 0 })
			Library:Tween(Items.Fade, { BackgroundTransparency = 1 })
		end

		Connect(Items.ArrowHolder.MouseButton1Click, function()
			arrowClick = true

			task.defer(function()
				arrowClick = false

			if arrowClick then
				return

			Cfg.Set(Cfg.Enabled)
		end)

		Connect(Items.Object.MouseButton1Click, function()
			Cfg.Enabled = not Cfg.Enabled
			Cfg.Set(Cfg.Enabled)
		end)

		Cfg.SetExpanded(false)
	end

	Cfg.Set(Cfg.Enabled)

	ConfigFlags[Cfg.Flag] = Cfg.Set

	if Cfg.Folding then
		task.defer(Cfg.RefreshFolding)
	end

	-- Metatable first so default Keybind can use Cfg:Keybind
	setmetatable(Cfg, Library)

	--[[
			Every Toggle gets an empty Keybind (no default key) with bidirectional Enabled sync.
			Pass Keybind = { Name, Flag, Mode, ... } only to override defaults; Keybind = false to skip.
			Defaults: Name = Toggle.Name, Flag = Toggle.Flag .. " Keybind", Mode = "Toggle", ShowInList = true.
		]]
	local keybindProp = properties.Keybind
	if keybindProp == false then
		-- Explicit opt-out: no chip, no list entry, no sync pair
	else
		if keybindProp == nil or keybindProp == true then
			keybindProp = {}
		end
		if Type(keybindProp) == "table" then
			local Ready = false
			local Syncing = false
			local kbProps = {
				Name = keybindProp.Name or Cfg.Name,
				Flag = keybindProp.Flag or (Cfg.Flag .. " Keybind"),
				Mode = keybindProp.Mode or "Toggle",
				ShowInList = keybindProp.ShowInList ~= false,
				Key = keybindProp.Key,
				Default = keybindProp.Default,
				Callback = keybindProp.Callback,
			}
			local OriginalCallback = kbProps.Callback

			kbProps.Callback = function(Active)
				if OriginalCallback then
					OriginalCallback(Active)
				end
				if Library.IsLoadingConfig then
					return
				end
				if not Ready or Syncing or Type(Active) ~= "boolean" then
					return
				end
				-- After loadcfg, Enabled may drift from Flag; still Set once so the master toggle matches keybind state
				Syncing = true
				Cfg.Set(Active)
				Syncing = false
			end

			local Keybind = Cfg:Keybind(kbProps)
			local originalSet = Cfg.Set
			Cfg.Set = function(bool)
				originalSet(bool)
				if Library.IsLoadingConfig then
					return
				end
				if not Ready or Syncing or Type(bool) ~= "boolean" then
					return
				end
				if Keybind.Active == bool then
					return
				end
				Syncing = true
				Keybind.Set(bool)
				Syncing = false
			end
			Ready = true
			Cfg.BoundKeybind = Keybind
			Insert(Library.BoundTogglePairs, {
				Toggle = Cfg,
				Keybind = Keybind,
			})
		end
	end

	return Cfg
end

	---Resize the scroll area to the visible row cap and, while open, the whole popup to match.
	function Cfg.ReflowOptions()
		local shown = #Cfg.OptionInstances
		local rows = Max(1, Min(shown, DROPDOWN_MAX_VISIBLE))
		Items.OptionScroll.Size = NewUDim2(1, 0, 0, rows * DROPDOWN_ROW_HEIGHT)
		Items.OptionScroll.ScrollBarThickness = shown > DROPDOWN_MAX_VISIBLE and 2 or 0
		Items.OptionScroll.CanvasPosition = Vector2.new(0, 0)
		if Items.DropdownElements.Visible and Items.DropdownElements.Parent == Library.Items then
			Items.DropdownElements.Size = UDim2.fromOffset(
				Items.Outline.AbsoluteSize.X + 1,
				rows * DROPDOWN_ROW_HEIGHT + DROPDOWN_SEARCH_HEIGHT + 8
			)
		end
	end

	function Cfg.ResetSearch()
		if Items.SearchBox and Items.SearchBox.Text ~= "" then
			Items.SearchBox.Text = ""
		end
	end

	---Filter Options by a plain, case-insensitive substring and re-render, keeping the selection highlight.
	function Cfg.ApplyOptionSearch()
		local raw = tostring(Items.SearchBox and Items.SearchBox.Text or "")
		local query = string.lower(raw):gsub("^%s+", ""):gsub("%s+$", "")
		local filtered = {}
		if query == "" then
			for _, option in Cfg.Options do
				filtered[#filtered + 1] = option
			end
		else
			for _, option in Cfg.Options do
				if string.find(string.lower(tostring(option)), query, 1, true) then
					filtered[#filtered + 1] = option
				end
			end
		end
		Cfg.RefreshOptions(filtered)
		if Cfg.Current ~= nil then
			Cfg.Set(Cfg.Current, true)
		end
	end

	function Cfg.SetVisible(bool)
		-- Only close a different open element when opening; closing self must not re-enter CloseElement
		if bool and Library.OpenElement ~= Cfg then
			Library:CloseElement()
		end

		Items.DropdownElements.Position =
			NewUDim2(0, Items.Outline.AbsolutePosition.X, 0, Items.Outline.AbsolutePosition.Y + 80)
		Items.DropdownElements.Size = UDim2.fromOffset(Items.Outline.AbsoluteSize.X + 1, 0)
		Items.DropdownElements.Visible = bool
		Items.DropdownElements.Parent = bool and Library.Items or Library.Other

		if bool then
			Cfg.ResetSearch()
			Cfg.ReflowOptions()
		end

		if not Cfg.Multi then
			Items.Icon.Text = bool and "+" or "-"
		end

		-- Cfg.Tween(bool)

		if bool then
			Library.OpenElement = Cfg
		elseif Library.OpenElement == Cfg then
			Library.OpenElement = {}
		end
	end

	function Cfg.Set(value, skipCallback)
		local Selected = {}
		local IsTable = Type(value) == "table"

		for _, option in Cfg.OptionInstances do
			if option.Text == value or (IsTable and table.find(value, option.Text)) then
				Insert(Selected, option.Text)
				Cfg.MultiItems = Selected
				option.TextColor3 = themes.preset.text_color
			else
				option.TextColor3 = themes.preset.text_muted
				option.BackgroundTransparency = 1
			end
		end

		Items.InnerText.Text = if IsTable then table.concat(Selected, ", ") else Selected[1] or ""
		Flags[Cfg.Flag] = if IsTable then Selected else Selected[1]
		Cfg.Current = Flags[Cfg.Flag]

		-- During config load, only refresh the Mode dropdown display; do not callback into Keybind.SetMode
		if not skipCallback then
			Cfg.Callback(Flags[Cfg.Flag])
		end
	end

	function Cfg.RefreshOptions(options)
		for _, option in Cfg.OptionInstances do
			option:Destroy()
		end

		Cfg.OptionInstances = {}

		for _, option in options do
			local Button = Cfg.RenderOption(option)

			Connect(Button.MouseButton1Down, function()
				if Cfg.Multi then
					local Selected = table.find(Cfg.MultiItems, Button.Text)

					if Selected then
						table.remove(Cfg.MultiItems, Selected)
					else
						Insert(Cfg.MultiItems, Button.Text)
					end

					Cfg.Set(Cfg.MultiItems)
				else
					Cfg.SetVisible(false)
					Cfg.Open = false

					Cfg.Set(Button.Text)
				end
			end)
		end

		Cfg.ReflowOptions()
	end

	function Cfg.Tween(bool)
		if Cfg.Tweening == true then
			return
		end

		Cfg.Tweening = true

		if bool then
			Items.DropdownElements.Visible = true
			Items.DropdownElements.Parent = Library.Items
		end

		local Children = GetDescendants(Items.DropdownElements)
		Insert(Children, Items.DropdownElements)

		local Tween
		for ci = 1, #Children do
			local obj = Children[ci]
			local Index = Library:GetTransparency(obj)

			if not Index then
				continue
			end

			if Type(Index) == "table" then
				for _, prop in Index do
					Tween = Library:Fade(obj, prop, bool, 0.1)
				end
			else
				Tween = Library:Fade(obj, Index, bool, 0.1)
			end
		end

		Delay(0.09, function()
			Cfg.Tweening = false
			Items.DropdownElements.Visible = bool
			Items.DropdownElements.Parent = bool and Library.Items or Library.Other
		end)
	end

	Connect(Items.Outline.MouseButton1Click, function()
		Cfg.Open = not Cfg.Open

		Cfg.SetVisible(Cfg.Open)
	end)

	Library:Connection(UserInputService.InputBegan, function(input, game_event)
		if isPressInput(input) then
			if Items.DropdownElements.Visible and not Library:Hovering({ Items.DropdownElements, Items.Dropdown }) then
				Cfg.SetVisible(false)
				Cfg.Open = false
			end
		end
	end)

	Flags[Cfg.Flag] = {}
	ConfigFlags[Cfg.Flag] = Cfg.Set

	Cfg.RefreshOptions(Cfg.Options)
	Cfg.Set(Cfg.Default)

	Connect(GetPropertyChangedSignal(Items.SearchBox, "Text"), function()
		Cfg.ApplyOptionSearch()
	end)

	return setmetatable(Cfg, Library)
end

	function Cfg.SetMode(mode)
		Cfg.Mode = mode

		-- During config load, change Mode only; do not Set(false) on Hold and turn the parent toggle off
		if not Library.IsLoadingConfig then
			if mode == "Always" then
				Cfg.Set(true)
			elseif mode == "Hold" then
				Cfg.Set(false)
			end
		end

		if Type(Flags[Cfg.Flag]) == "table" then
			Flags[Cfg.Flag].Mode = mode
			Flags[Cfg.Flag].mode = mode
		end
	end

	function Cfg.Set(input)
		if Type(input) == "boolean" then
			Cfg.Active = input

			if Cfg.Mode == "Always" then
				Cfg.Active = true
			end
		elseif TypeOf(input) == "EnumItem" or (Type(input) == "string" and string.find(input, "Enum", 1, true)) then
			if TypeOf(input) == "string" then
				input = Library:ConvertEnum(input)
			end
			input = (TypeOf(input) == "EnumItem" and input.Name == "Escape") and "NONE" or input
			Cfg.Key = input or "NONE"
		elseif table.find({ "Toggle", "Hold", "Always" }, input) then
			if input == "Always" then
				Cfg.Active = true
			end

			Cfg.Mode = input
			Cfg.SetMode(Cfg.Mode)
		elseif Type(input) == "table" then
			local mode = input.Mode or input.mode
			local active = input.Active
			if active == nil then
				active = input.active
			end
			local key = input.Key or input.key
			if Type(key) == "string" and key ~= "NONE" and key ~= "nil" then
				key = Library:ConvertEnum(key)
			end
			if key == Enum.KeyCode.Escape or key == nil or key == "nil" or key == "NONE" then
				key = "NONE"
			end

			Cfg.Key = key or "NONE"
			Cfg.Mode = mode or Cfg.Mode or "Toggle"
			if Cfg.Mode == "Always" then
				Cfg.Active = true
			elseif active ~= nil then
				Cfg.Active = active == true
			end

			-- Write Key/Mode/Active directly; do not recurse through SetMode which forces Hold=false
			if Items.Dropdown and Items.Dropdown.Set then
				Items.Dropdown.Set(Cfg.Mode, true)
			end
		end

		Cfg.Callback(Cfg.Active)

		local text = (tostring(Cfg.Key) ~= "Enums" and (Keys[Cfg.Key] or tostring(Cfg.Key):gsub("Enum.", "")) or nil)
		local __text = text and tostring(text):gsub("KeyCode.", ""):gsub("UserInputType.", "")

		Items.Key.Text = __text

		if Items.Keybinds then
			local hasValue = __text ~= nil and __text ~= "" and __text ~= "NONE" and __text ~= "nil"
			if Items.KeybindsRow then
				Items.KeybindsRow.Visible = hasValue
			else
				Items.Keybinds.Visible = hasValue
			end
			if hasValue then
				local listText = Format("[%s] %s", __text, Cfg.Name or Cfg.Flag or "Key")
				Items.Keybinds.Text = listText
				Items.Keybinds.TextColor3 = Cfg.Active and themes.preset.accent or themes.preset.text_color
				if Items.KeybindsRow then
					Items.KeybindsRow.Size =
						NewUDim2(0, getOverlayTextWidth(listText, NOTIFY_WIDTH_PADDING), 0, NOTIFY_HEIGHT)
				end
			end
		end

		Flags[Cfg.Flag] = {
			Mode = Cfg.Mode,
			Key = Cfg.Key,
			Active = Cfg.Active,
			mode = Cfg.Mode,
			key = Cfg.Key,
			active = Cfg.Active,
		}
	end

	---Toggle Mode panel visibility with a fade mask so content does not pop in instantly.
	---@param bool boolean Whether the panel is shown
	function Cfg.SetVisible(bool)
		Items.ModeHolder.Position =
			NewUDim2(0, Items.KeybindOutline.AbsolutePosition.X + 2, 0, Items.KeybindOutline.AbsolutePosition.Y + 74)

		if bool then
			Items.ModeHolder.Visible = true
			Items.Fade.BackgroundTransparency = 0
			-- Fade the panel in via the mask; default TweenInfo reads Settings animation config.
			Library:Tween(Items.Fade, { BackgroundTransparency = 1 })
			return
		end

		-- Finish the mask fade-out before hiding the panel so closing does not drop content abruptly.
		local Tween = Library:Tween(Items.Fade, { BackgroundTransparency = 0 })
		Library:Connection(Tween.Completed, function()
			if not Cfg.Open then
				Items.ModeHolder.Visible = false
			end
		end)
	end

	Connect(Items.KeybindOutline.MouseButton1Down, function()
		Wait()
		Items.Key.Text = "..."

		Cfg.Binding = Library:Connection(UserInputService.InputBegan, function(keycode, game_event)
			Cfg.Set(keycode.KeyCode ~= Enum.KeyCode.Unknown and keycode.KeyCode or keycode.UserInputType)

			Disconnect(Cfg.Binding)
			Cfg.Binding = nil
		end)
	end)

	Connect(Items.KeybindOutline.MouseButton2Down, function()
		Cfg.Open = not Cfg.Open

		Cfg.SetVisible(Cfg.Open)
	end)

	Library:Connection(UserInputService.InputBegan, function(input, game_event)
		if isPressInput(input) then
			if
				(Items.Dropdown.Items.DropdownElements.Visible and Items.ModeHolder.Visible)
				and not (Library:Hovering(Items.Dropdown.Items.DropdownElements) or Library:Hovering(Items.ModeHolder))
			then
				Items.Dropdown.SetVisible(false)
				Items.Dropdown.Visible = false

				Cfg.SetVisible(false)
				Cfg.Open = false
			end
		end

		if not game_event then
			local selected_key = input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode
				or input.UserInputType

			if selected_key == Cfg.Key then
				if Cfg.Mode == "Toggle" then
					Cfg.Active = not Cfg.Active
					Cfg.Set(Cfg.Active)
				elseif Cfg.Mode == "Hold" then
					Cfg.Set(true)
				end
			end
		end
	end)

	Library:Connection(UserInputService.InputEnded, function(input, game_event)
		if game_event then
			return
		end

		local selected_key = input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode or input.UserInputType

		if selected_key == Cfg.Key then
			if Cfg.Mode == "Hold" then
				Cfg.Set(false)
			end
		end
	end)

	Cfg.Set({ Mode = Cfg.Mode, Active = Cfg.Active, Key = Cfg.Key })
	ConfigFlags[Cfg.Flag] = Cfg.Set
	Items.Dropdown.Set(Cfg.Mode, true)

	return setmetatable(Cfg, Library)
end

-- Notification Library
--[[
		Tip visuals follow constant pf.lua Card: inline fill + outline stroke + left accent + gradient body.
		Tips stack under the Notification chip. Fading tips keep their screen position frozen but still
		reserve stack height until destroyed. Layout reflow uses ScheduleRefresh to coalesce same-frame
		updates when multiple tips expire together. Stack origin flips at the viewport midpoint.
		Fade uses 0.15s Linear.
	]]
local NOTIFY_STACK_ITEM_GAP = 2
local NOTIFY_MAX_VISIBLE = Notifications.MaxVisible or 25
local NOTIFY_FADE_INFO = TweenInfo.new(0.15, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)

---Measure tip width using constant pf.lua Card padding rules.
---@param text string Notification body text
---@return number Outline width in pixels
local function getNotificationWidth(text)
	text = text or ""
	local ok, bounds = pcall(function()
		local params = NewInstance("GetTextBoundsParams")
		params.Text = text
		params.Font = Library.Font
		params.Size = Library.TextSize
		params.Width = 10000
		return TextService:GetTextBoundsAsync(params)
	end)
	if ok and bounds then
		return bounds.X + NOTIFY_WIDTH_PADDING
	end
	local legacyBounds =
		TextService:GetTextSize(text, Library.TextSize, Enum.Font.SourceSans, NewVector2(10000, NOTIFY_HEIGHT))
	return legacyBounds.X + NOTIFY_WIDTH_PADDING
end

---Resolve tip-stack layout under the Notification chip.
---@return number, number, Vector2 Origin position and card anchor point
local function getNotificationLayout()
	local anchor = Library.NotifyAnchor
	if anchor and anchor.Parent then
		local gap = Library.NotifyStackGap or 2
		local anchorPos = anchor.Position
		local viewport = Camera.ViewportSize
		local isRightSide = anchor.AbsolutePosition.X + anchor.AbsoluteSize.X / 2 >= viewport.X / 2
		local anchorPoint = isRightSide and NewVector2(1, 0) or NewVector2(0, 0)
		local originX = anchorPos.X.Offset
		if isRightSide then
			originX += anchor.AbsoluteSize.X
		end
		return originX, anchorPos.Y.Offset + anchor.AbsoluteSize.Y + gap, anchorPoint
	end
	return 20, 50, NewVector2(0, 0)
end

---Re-layout active tips from the Notification chip without tweened reflow (avoids overlap).
---Fading tips keep their current Position but still consume stack height.
---@return number Next Y below the stack
function Notifications:RefreshNotifications()
	local originX, offsetY, anchorPoint = getNotificationLayout()

	-- Fading tips still occupy a slot so new tips cannot reuse their old coordinates.
	for i = 1, #Notifications.Notifs do
		local v = Notifications.Notifs[i]
		if v and v.Parent then
			local itemHeight = v.AbsoluteSize.Y + NOTIFY_STACK_ITEM_GAP
			if not Notifications.Fading[v] then
				-- Stack reflow is layout only, not a user-facing transition.
				v.AnchorPoint = anchorPoint
				v.Position = UDim2.fromOffset(originX, offsetY)
			end
			offsetY += itemHeight
		end
	end

	return offsetY
end

---Coalesce layout reflows to one deferred pass per frame (batch fade-out completions).
function Notifications:ScheduleRefresh()
	if self.RefreshScheduled then
		return
	end
	self.RefreshScheduled = true
	task.defer(function()
		self.RefreshScheduled = false
		self:RefreshNotifications()
	end)
end

---Cancel the lifetime progress tween for a tip so fade-out does not fight position motion.
---@param path Frame Tip outline
local function cancelNotificationProgressTween(path)
	local progressTween = Notifications.ProgressTweens[path]
	if progressTween then
		progressTween:Cancel()
		Notifications.ProgressTweens[path] = nil
	end
end

---Fade out and drop a tip from the active stack while keeping its layout slot during the fade.
---@param path Frame Tip outline
function Notifications:Remove(path)
	if not path or Notifications.Fading[path] then
		return
	end

	local tracked = false
	for i = 1, #Notifications.Notifs do
		local notif = Notifications.Notifs[i]
		if notif == path then
			tracked = true
			break
		end
	end
	if not tracked then
		return
	end

	-- Mark fading first so the tip keeps its stack slot until the tween finishes.
	Notifications.Fading[path] = true
	cancelNotificationProgressTween(path)
	Notifications:FadeNotifs(path, true, function()
		Notifications.Fading[path] = nil
		cancelNotificationProgressTween(path)
		for index = 1, #Notifications.Notifs do
			local notif = Notifications.Notifs[index]
			if notif == path then
				table.remove(Notifications.Notifs, index)
				break
			end
		end
		if path and path.Parent then
			path:Destroy()
		end
		-- After destroy, pull remaining active tips upward (coalesced when several fade together).
		Notifications:ScheduleRefresh()
	end)
end

---Shrink the active tip count to maxVisible, preferring the oldest non-fading tip.
---@param maxVisible number Max tips to keep (may be 0)
function Notifications:EnforceLimit(maxVisible)
	maxVisible = Max(0, Floor(tonumber(maxVisible) or Notifications.MaxVisible or NOTIFY_MAX_VISIBLE))

	while true do
		local activeCount = 0
		local oldestActive
		for i = 1, #Notifications.Notifs do
			local notif = Notifications.Notifs[i]
			if not Notifications.Fading[notif] and notif and notif.Parent then
				activeCount += 1
				oldestActive = oldestActive or notif
			end
		end
		if activeCount <= maxVisible or not oldestActive then
			break
		end
		-- Fading tips remain in Notifs; only pick among still-active ones.
		Notifications:Remove(oldestActive)
	end
end

---Tween transparency for a tip and all descendants; run on_completed when the outline finishes.
---@param path Frame Tip outline
---@param is_fading boolean True to fade out
---@param on_completed function|nil Callback after the outline tween completes
---@return Tween Outline tween
function Notifications:FadeNotifs(path, is_fading, on_completed)
	local fading = is_fading and 1 or 0
	-- Tip fade timing matches constant pf.lua: 0.15s Linear.
	local fadeInfo = NOTIFY_FADE_INFO
	local tweens = {}

	local function createTween(instance, properties)
		local tween = TweenService:Create(instance, fadeInfo, properties)
		tweens[#tweens + 1] = tween
		return tween
	end

	local tween = createTween(path, { BackgroundTransparency = fading })
	local fadeDescendants = GetDescendants(path)
	for di = 1, #fadeDescendants do
		local instance = fadeDescendants[di]
		if not IsA(instance, "GuiObject") then
			if IsA(instance, "UIStroke") then
				createTween(instance, { Transparency = fading })
			end
			continue
		end

		if IsA(instance, "TextLabel") then
			createTween(instance, { TextTransparency = fading })
		elseif IsA(instance, "Frame") then
			createTween(instance, { BackgroundTransparency = fading })
		end
	end

	if on_completed then
		Connect(tween.Completed, on_completed)
	end

	-- Connect every tween before Play so short fades still fire Completed.
	for i = 1, #tweens do
		tweens[i]:Play()
	end
	return tween
end

---Create a tip with a left accent and a bottom lifetime progress bar.
---@param properties table Tip name and lifetime
---@return table Tip config
function Notifications:Create(properties)
	local Cfg = {
		Name = properties.Name or "This is a title!",
		Lifetime = properties.LifeTime or 3,

		Items = {},
		outline,
	}

	-- Free a slot before inserting so the stack never exceeds MaxVisible.
	Notifications:EnforceLimit(Max(0, (Notifications.MaxVisible or NOTIFY_MAX_VISIBLE) - 1))

	local originX, originY, anchorPoint = getNotificationLayout()
	local notifyText = Cfg.Name
	local notifyWidth = getNotificationWidth(notifyText)
	local progressWidth = Floor(notifyWidth / 10.36)
	local Items = Cfg.Items
	do
		local card = CreateNotifyStyleCard({
			Parent = Library.Items,
			Text = notifyText,
			AnchorPoint = anchorPoint,
			Position = NewUDim2(0, originX, 0, originY),
			Width = notifyWidth,
		})
		Items.Outline = card.Outline
		Items.OutlineStroke = card.Stroke
		Items.Holder = card.Holder
		Items.Accent = card.Accent
		Items.Text = card.Text

		-- Bottom progress bar: fixed narrow width, slides left-to-right over Lifetime.
		Items.AccentLine = Library:Create("Frame", {
			Parent = Items.Outline,
			Name = "\0",
			Position = NewUDim2(0, 0, 1, -1),
			BorderColor3 = NewRGB(0, 0, 0),
			Size = NewUDim2(0, progressWidth, 0, 1),
			BorderSizePixel = 0,
			ZIndex = 100,
			BackgroundColor3 = themes.preset.accent,
		})
		Library:Themify(Items.AccentLine, "accent", "BackgroundColor3")
	end

	Notifications.Notifs[#Notifications.Notifs + 1] = Items.Outline

	-- Place the new tip at its final stack slot immediately (no slide-from-origin overlap).
	do
		local originX, originY, anchorPoint = getNotificationLayout()
		local offsetY = originY
		for i = 1, #Notifications.Notifs do
			local above = Notifications.Notifs[i]
			if above ~= Items.Outline then
				offsetY += above.AbsoluteSize.Y + NOTIFY_STACK_ITEM_GAP
			end
		end
		Items.Outline.AnchorPoint = anchorPoint
		Items.Outline.Position = UDim2.fromOffset(originX, offsetY)
	end

	Notifications:RefreshNotifications()

	-- Fade only transparency; keep the precomputed stack position.
	Items.Outline.BackgroundTransparency = 1
	local outlineDescendants = GetDescendants(Items.Outline)
	for di = 1, #outlineDescendants do
		local instance = outlineDescendants[di]
		if not IsA(instance, "GuiObject") then
			if IsA(instance, "UIStroke") then
				instance.Transparency = 1
			end
		elseif IsA(instance, "TextLabel") then
			instance.TextTransparency = 1
		elseif IsA(instance, "Frame") then
			instance.BackgroundTransparency = 1
		end
	end
	Notifications:FadeNotifs(Items.Outline, false)

	-- Progress bar motion matches constant pf.lua (Linear over Lifetime).
	Notifications.ProgressTweens[Items.Outline] = Library:Tween(
		Items.AccentLine,
		{ Position = NewUDim2(1, -progressWidth, 1, -1) },
		TweenInfo.new(Cfg.Lifetime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out, 0, false, 0)
	)

	Delay(Cfg.Lifetime, function()
		Notifications:Remove(Items.Outline)
	end)
end

return Library, Notifications, themes
